<script>
class Game2048 {
  constructor() {
    // 新游戏，不保存游戏状态，与其他游戏保持一致
    this.size = 4;
    this.board = [];
    this.score = 0;
    this.bestScore = localStorage.getItem('2048-best-score') ? parseInt(localStorage.getItem('2048-best-score')) : 0;
    this.won = false;
    this.over = false;
    this.keepPlaying = false; // 拼出 2048 后选择继续挑战
    this.gameBoardElement = null;
    this.nextTileId = 1; // 方块唯一 id，自增分配，跨 render 稳定追踪
    this.tileElements = {}; // id -> 方块 DOM 元素，用于增量更新
    this.layout = null; // 当前渲染布局 { boardSize, tileSize, gap }
    this.mergedTiles = []; // 最近一次移动产生的合并结果（棋盘坐标）
    this.lastMovements = []; // 最近一次移动的方块位置映射 { id, from, to }
    this.snapshot = null; // 撤销快照 { board, score }，仅保留最近一步
    this.lastMoveTime = 0; // 输入节流时间戳
    this.renderVersion = 0; // 渲染版本号，用于作废延迟动画回调
    this.initBoard();
    this.setupEventListeners();
    this.render();
    this.updateScoreDisplay();
  }

  initBoard() {
    this.board = Array(this.size).fill().map(() => Array(this.size).fill(null));
    this.addRandomTile();
    this.addRandomTile();
  }

  cloneBoard() {
    return this.board.map(row => row.map(cell => (cell ? { id: cell.id, value: cell.value } : null)));
  }

  addRandomTile() {
    const emptyCells = [];
    for (let i = 0; i < this.size; i++) {
      for (let j = 0; j < this.size; j++) {
        if (!this.board[i][j]) {
          emptyCells.push({ i, j });
        }
      }
    }

    if (emptyCells.length === 0) return null;

    const { i, j } = emptyCells[Math.floor(Math.random() * emptyCells.length)];
    const tile = { id: this.nextTileId++, value: Math.random() < 0.9 ? 2 : 4 };
    this.board[i][j] = tile;
    return { id: tile.id, value: tile.value, i, j };
  }

  move(direction) {
    if (this.over || this.won) return; // 游戏结束或胜利弹窗展示中时不允许移动

    // 输入节流，防止连续快速按键导致动画错乱
    const now = Date.now();
    if (now - this.lastMoveTime < 110) return;

    // 移动前先保存撤销快照（仅保留最近一步）
    const snapshot = { board: this.cloneBoard(), score: this.score };

    let moved = false;

    if (direction === 'left') {
      moved = this.moveLeft();
    } else if (direction === 'right') {
      moved = this.moveRight();
    } else if (direction === 'up') {
      moved = this.moveUp();
    } else if (direction === 'down') {
      moved = this.moveDown();
    }

    if (moved) {
      this.snapshot = snapshot;
      this.lastMoveTime = now;
      const newTile = this.addRandomTile();
      this.renderMove(newTile);
      this.updateScoreDisplay();

      // 使用正确的游戏结束检查
      this.checkGameOver();
    }
  }

  undo() {
    if (!this.snapshot) return;

    this.board = this.snapshot.board;
    this.score = this.snapshot.score;
    this.snapshot = null; // 不可连续撤销
    this.over = false;
    this.won = false;
    this.hideGameOver();
    const wonElement = document.getElementById('game-won');
    if (wonElement) wonElement.remove();
    this.updateScoreDisplay();
    this.render();
  }

  checkGameOver() {
    // 首先检查是否获胜（选择继续挑战后不再弹胜利窗）
    if (!this.keepPlaying && this.hasWon()) {
      this.won = true;
      this.showGameWon();
      return;
    }

    // 没有空位且无可合并方块时游戏结束
    if (this.isGameOver()) {
      this.over = true;
      this.showGameOver();
    }
  }

  // 将一行方块（null 或 {id, value}）向索引 0 方向滑动、合并
  // 返回新行、每个方块的位置映射（行内索引）以及合并记录
  slideLine(line) {
    const entries = [];
    for (let idx = 0; idx < line.length; idx++) {
      if (line[idx]) {
        entries.push({ id: line[idx].id, value: line[idx].value, from: idx });
      }
    }

    const result = [];
    const movements = [];
    const merges = [];
    let k = 0;
    while (k < entries.length) {
      const to = result.length;
      if (k + 1 < entries.length && entries[k].value === entries[k + 1].value) {
        const merged = { id: this.nextTileId++, value: entries[k].value * 2 };
        result.push(merged);
        movements.push({ id: entries[k].id, from: entries[k].from, to });
        movements.push({ id: entries[k + 1].id, from: entries[k + 1].from, to });
        merges.push({ index: to, value: merged.value, id: merged.id, sourceIds: [entries[k].id, entries[k + 1].id] });
        this.score += merged.value;
        k += 2;
      } else {
        result.push({ id: entries[k].id, value: entries[k].value });
        movements.push({ id: entries[k].id, from: entries[k].from, to });
        k += 1;
      }
    }

    while (result.length < this.size) {
      result.push(null);
    }

    const moved = movements.some(m => m.from !== m.to);
    return { moved, result, movements, merges };
  }

  moveLeft() {
    this.mergedTiles = [];
    this.lastMovements = [];
    let moved = false;

    for (let i = 0; i < this.size; i++) {
      const r = this.slideLine(this.board[i]);
      this.board[i] = r.result;
      moved = moved || r.moved;
      r.movements.forEach(m => this.lastMovements.push({ id: m.id, from: { i, j: m.from }, to: { i, j: m.to } }));
      r.merges.forEach(mg => this.mergedTiles.push({ i, j: mg.index, value: mg.value, id: mg.id, sourceIds: mg.sourceIds }));
    }

    return moved;
  }

  moveRight() {
    this.mergedTiles = [];
    this.lastMovements = [];
    let moved = false;

    for (let i = 0; i < this.size; i++) {
      // 逆序取出行，滑动后再映射回原坐标
      const line = [];
      for (let j = this.size - 1; j >= 0; j--) {
        line.push(this.board[i][j]);
      }
      const r = this.slideLine(line);
      for (let j = 0; j < this.size; j++) {
        this.board[i][this.size - 1 - j] = r.result[j];
      }
      moved = moved || r.moved;
      r.movements.forEach(m => this.lastMovements.push({
        id: m.id,
        from: { i, j: this.size - 1 - m.from },
        to: { i, j: this.size - 1 - m.to }
      }));
      r.merges.forEach(mg => this.mergedTiles.push({ i, j: this.size - 1 - mg.index, value: mg.value, id: mg.id, sourceIds: mg.sourceIds }));
    }

    return moved;
  }

  moveUp() {
    this.mergedTiles = [];
    this.lastMovements = [];
    let moved = false;

    for (let j = 0; j < this.size; j++) {
      const line = [];
      for (let i = 0; i < this.size; i++) {
        line.push(this.board[i][j]);
      }
      const r = this.slideLine(line);
      for (let i = 0; i < this.size; i++) {
        this.board[i][j] = r.result[i];
      }
      moved = moved || r.moved;
      r.movements.forEach(m => this.lastMovements.push({
        id: m.id,
        from: { i: m.from, j },
        to: { i: m.to, j }
      }));
      r.merges.forEach(mg => this.mergedTiles.push({ i: mg.index, j, value: mg.value, id: mg.id, sourceIds: mg.sourceIds }));
    }

    return moved;
  }

  moveDown() {
    this.mergedTiles = [];
    this.lastMovements = [];
    let moved = false;

    for (let j = 0; j < this.size; j++) {
      // 逆序取出列，滑动后再映射回原坐标
      const line = [];
      for (let i = this.size - 1; i >= 0; i--) {
        line.push(this.board[i][j]);
      }
      const r = this.slideLine(line);
      for (let i = 0; i < this.size; i++) {
        this.board[this.size - 1 - i][j] = r.result[i];
      }
      moved = moved || r.moved;
      r.movements.forEach(m => this.lastMovements.push({
        id: m.id,
        from: { i: this.size - 1 - m.from, j },
        to: { i: this.size - 1 - m.to, j }
      }));
      r.merges.forEach(mg => this.mergedTiles.push({ i: this.size - 1 - mg.index, j, value: mg.value, id: mg.id, sourceIds: mg.sourceIds }));
    }

    return moved;
  }

  isGameOver() {
    // 检查是否有空位
    for (let i = 0; i < this.size; i++) {
      for (let j = 0; j < this.size; j++) {
        if (!this.board[i][j]) {
          return false; // 有空位，游戏未结束
        }
      }
    }

    // 检查相邻方向是否存在相同数字
    for (let i = 0; i < this.size; i++) {
      for (let j = 0; j < this.size; j++) {
        const current = this.board[i][j].value;

        // 检查右侧
        if (j < this.size - 1 && this.board[i][j + 1].value === current) {
          return false; // 右侧有相同数字，游戏未结束
        }

        // 检查下方
        if (i < this.size - 1 && this.board[i + 1][j].value === current) {
          return false; // 下方有相同数字，游戏未结束
        }
      }
    }

    return true; // 没有空位且无可合并的相邻数字，游戏结束
  }

  showGameOver() {
    const gameOver = document.getElementById('game-over');
    if (gameOver) gameOver.classList.remove('hidden');
  }
  
  showGameWon() {
    // 如果胜利遮罩已存在，不重复创建
    if (document.getElementById('game-won')) return;

    const gameWonElement = document.createElement('div');
    gameWonElement.id = 'game-won';
    gameWonElement.className = '';
    gameWonElement.innerHTML = `
      <div class='game-over-content'>
        <div class='game-over-text'>恭喜你，拼出 2048！</div>
        <div class='won-buttons'>
          <button id='continue-button-won'>继续挑战</button>
          <button id='restart-button-won'>再玩一次</button>
        </div>
      </div>
    `;
    document.getElementById('game2048').appendChild(gameWonElement);

    // 继续挑战：关闭遮罩，之后不再因 2048 弹窗，可继续合成 4096/8192
    const continueButtonWon = document.getElementById('continue-button-won');
    if (continueButtonWon) {
      continueButtonWon.addEventListener('click', () => {
        this.keepPlaying = true;
        this.won = false;
        gameWonElement.remove();
      });
    }

    // 再玩一次
    const restartButtonWon = document.getElementById('restart-button-won');
    if (restartButtonWon) {
      restartButtonWon.addEventListener('click', () => {
        this.restart();
        gameWonElement.remove();
      });
    }
  }

  hasWon() {
    // 检查是否有 2048 及以上方块
    for (let i = 0; i < this.size; i++) {
      for (let j = 0; j < this.size; j++) {
        if (this.board[i][j] && this.board[i][j].value >= 2048) {
          return true;
        }
      }
    }
    return false;
  }

  hideGameOver() {
    const gameOver = document.getElementById('game-over');
    if (gameOver) gameOver.classList.add('hidden');
  }

  restart() {
    this.score = 0;
    this.won = false;
    this.over = false;
    this.keepPlaying = false;
    this.snapshot = null;
    this.mergedTiles = [];
    this.lastMovements = [];
    this.initBoard();
    this.hideGameOver();
    const wonElement = document.getElementById('game-won');
    if (wonElement) wonElement.remove();
    this.updateScoreDisplay();
    this.render();
  }

  updateScoreDisplay() {
    const scoreElement = document.getElementById('score');
    const bestScoreElement = document.getElementById('best') || document.getElementById('best-score');
    if (scoreElement) scoreElement.textContent = this.score;
    if (bestScoreElement) bestScoreElement.textContent = this.bestScore;

    // 更新最佳分数到本地存储
    if (this.score > this.bestScore) {
      this.bestScore = this.score;
      localStorage.setItem('2048-best-score', this.bestScore);
      if (bestScoreElement) bestScoreElement.textContent = this.bestScore;
    }
  }
  
  // 计算棋盘布局尺寸（随屏幕宽度自适应）
  computeLayout() {
    // 更精确的移动端适配，解决棋盘超出界面问题
    let boardSize;
    const screenWidth = window.innerWidth;
    const gameContainer = document.getElementById('game2048');

    // 获取游戏容器的可用宽度（在 DOM 可用时使用实际容器宽度）
    let availableWidth = screenWidth;
    if (gameContainer) {
      const containerRect = gameContainer.getBoundingClientRect();
      availableWidth = Math.min(screenWidth, containerRect.width);
    }

    // 根据可用宽度计算棋盘大小，增加更大的安全边距
    if (availableWidth < 350) {
      boardSize = availableWidth - 50; // 极小屏幕，增加边距
    } else if (availableWidth < 450) {
      boardSize = availableWidth - 70; // 小屏幕
    } else if (availableWidth < 550) {
      boardSize = availableWidth - 90; // 中等屏幕
    } else {
      boardSize = 400; // 桌面端固定大小
    }

    // 严格限制最大和最小尺寸
    boardSize = Math.min(boardSize, 450);
    boardSize = Math.max(boardSize, 200);

    const gap = 10;
    const tileSize = (boardSize - (gap * (this.size + 1))) / this.size;
    return { boardSize, tileSize, gap };
  }

  getTileColors(value) {
    const tileColors = {
      2: '#eee4da',
      4: '#ede0c8',
      8: '#f2b179',
      16: '#f59563',
      32: '#f67c5f',
      64: '#f65e3b',
      128: '#edcf72',
      256: '#edcc61',
      512: '#edc850',
      1024: '#edc53f',
      2048: '#edc22e'
    };

    if (value > 2048) {
      return { bg: '#3c3a32', fg: 'white' };
    }
    return {
      bg: tileColors[value] || tileColors[2],
      fg: value <= 4 ? '#776e65' : 'white'
    };
  }

  // 字号随方块尺寸缩放，保证小屏和桌面比例协调
  getTileFontSize(value, tileSize) {
    const digits = String(value).length;
    if (digits <= 2) return tileSize * 0.45;
    if (digits === 3) return tileSize * 0.38;
    return tileSize * 0.3;
  }

  createTileElement(id, value, i, j, layout) {
    const { tileSize, gap } = layout;
    const colors = this.getTileColors(value);
    const tile = document.createElement('div');
    tile.className = 'tile';
    tile.style.position = 'absolute';
    tile.style.left = `${gap + j * (tileSize + gap)}px`;
    tile.style.top = `${gap + i * (tileSize + gap)}px`;
    tile.style.width = `${tileSize}px`;
    tile.style.height = `${tileSize}px`;
    tile.style.backgroundColor = colors.bg;
    tile.style.color = colors.fg;
    tile.style.borderRadius = '8px';
    tile.style.display = 'flex';
    tile.style.justifyContent = 'center';
    tile.style.alignItems = 'center';
    tile.style.fontWeight = 'bold';
    tile.style.fontSize = `${this.getTileFontSize(value, tileSize)}px`;
    // 平移滑动动画：位置变化时平滑过渡
    tile.style.transition = 'left 0.1s ease-in-out, top 0.1s ease-in-out';
    tile.style.zIndex = '10';
    tile.textContent = value;
    return tile;
  }

  // 全量渲染（初始化 / 重新开始 / 撤销 / 页面恢复显示时使用）
  render() {
    this.renderVersion++;
    this.gameBoardElement = document.getElementById('game-board-2048');
    if (!this.gameBoardElement) return;

    const layout = this.computeLayout();
    this.layout = layout;
    const { boardSize, tileSize, gap } = layout;
    const gameContainer = document.getElementById('game2048');

    this.gameBoardElement.innerHTML = '';
    this.tileElements = {};

    // 设置棋盘样式
    this.gameBoardElement.style.width = `${boardSize}px`;
    this.gameBoardElement.style.height = `${boardSize}px`;
    this.gameBoardElement.style.position = 'relative';
    this.gameBoardElement.style.backgroundColor = '#bbada0';
    this.gameBoardElement.style.borderRadius = '8px';
    this.gameBoardElement.style.margin = '0 auto';
    this.gameBoardElement.style.boxSizing = 'border-box';
    this.gameBoardElement.style.maxWidth = '100%';
    this.gameBoardElement.style.padding = '0';
    this.gameBoardElement.style.display = 'block';

    // 设置游戏容器样式
    if (gameContainer) {
      gameContainer.style.width = '100%';
      gameContainer.style.maxWidth = '100vw';
      gameContainer.style.overflow = 'hidden';
      gameContainer.style.boxSizing = 'border-box';
    }

    // 渲染空格子（背景格子）
    for (let i = 0; i < this.size; i++) {
      for (let j = 0; j < this.size; j++) {
        const emptyTile = document.createElement('div');
        emptyTile.className = 'empty-tile';
        emptyTile.style.position = 'absolute';
        emptyTile.style.left = `${gap + j * (tileSize + gap)}px`;
        emptyTile.style.top = `${gap + i * (tileSize + gap)}px`;
        emptyTile.style.width = `${tileSize}px`;
        emptyTile.style.height = `${tileSize}px`;
        emptyTile.style.backgroundColor = '#cdc1b4';
        emptyTile.style.borderRadius = '8px';
        emptyTile.style.zIndex = '1';
        this.gameBoardElement.appendChild(emptyTile);
      }
    }

    // 渲染数字方块
    for (let i = 0; i < this.size; i++) {
      for (let j = 0; j < this.size; j++) {
        const cell = this.board[i][j];
        if (cell) {
          const tile = this.createTileElement(cell.id, cell.value, i, j, layout);
          this.tileElements[cell.id] = tile;
          this.gameBoardElement.appendChild(tile);
        }
      }
    }
  }

  // 移动后的增量渲染：已存在方块通过 CSS transition 滑动到新位置，
  // 合并结果在滑动到位后播放合并弹跳动画，新随机方块播放淡入缩放动画
  renderMove(newTile) {
    this.gameBoardElement = document.getElementById('game-board-2048');
    if (!this.gameBoardElement) return;

    const layout = this.computeLayout();

    // 布局尺寸变化（如窗口缩放、容器重排）时退化为全量重绘
    if (!this.layout || Math.abs(this.layout.tileSize - layout.tileSize) > 0.5) {
      this.render();
      return;
    }

    this.renderVersion++;
    const version = this.renderVersion;
    const { tileSize, gap } = layout;
    const merges = this.mergedTiles.slice();
    const boardElement = this.gameBoardElement;

    // 滑动所有已存在方块到新位置（CSS transition 平滑过渡）
    this.lastMovements.forEach(m => {
      const el = this.tileElements[m.id];
      if (!el) return;
      el.style.left = `${gap + m.to.j * (tileSize + gap)}px`;
      el.style.top = `${gap + m.to.i * (tileSize + gap)}px`;
    });

    // 滑动到位后再处理合并结果与新方块出现
    setTimeout(() => {
      if (version !== this.renderVersion) return; // 期间发生了重绘/撤销/重开，放弃本次动画收尾

      merges.forEach(mg => {
        // 移除合并源方块
        mg.sourceIds.forEach(id => {
          const el = this.tileElements[id];
          if (el) {
            el.remove();
            delete this.tileElements[id];
          }
        });
        // 创建合并后的新方块并播放合并弹跳动画
        const mergedEl = this.createTileElement(mg.id, mg.value, mg.i, mg.j, layout);
        mergedEl.style.animation = 'collisionMergeEffect 0.18s ease-out';
        mergedEl.style.zIndex = '100';
        this.tileElements[mg.id] = mergedEl;
        boardElement.appendChild(mergedEl);
      });

      // 新随机方块淡入缩放出现
      if (newTile) {
        const newEl = this.createTileElement(newTile.id, newTile.value, newTile.i, newTile.j, layout);
        newEl.style.animation = 'appearEffect 0.12s ease-out';
        this.tileElements[newTile.id] = newEl;
        boardElement.appendChild(newEl);
      }
    }, 100);
  }

  setupEventListeners() {
    const handleKeyDown = (e) => {
      switch(e.key) {
        case 'ArrowUp':
        case 'w':
        case 'W':
          e.preventDefault();
          this.move('up');
          break;
        case 'ArrowDown':
        case 's':
        case 'S':
          e.preventDefault();
          this.move('down');
          break;
        case 'ArrowLeft':
        case 'a':
        case 'A':
          e.preventDefault();
          this.move('left');
          break;
        case 'ArrowRight':
        case 'd':
        case 'D':
          e.preventDefault();
          this.move('right');
          break;
      }
    };
    document.addEventListener('keydown', handleKeyDown);
    
    let startX, startY;
    let touchStartedOnBoard = false; // 记录触摸是否开始于棋盘区域
    const SWIPE_THRESHOLD = 30; // 滑动阈值，防止点击触发
    
    const handleTouchStart = (e) => {
      const touch = e.touches[0];
      startX = touch.clientX;
      startY = touch.clientY;
      
      // 检查触摸点是否在棋盘区域内
      const gameBoardElement = document.getElementById('game-board-2048');
      if (gameBoardElement) {
        const rect = gameBoardElement.getBoundingClientRect();
        touchStartedOnBoard = touch.clientX >= rect.left && 
                             touch.clientX <= rect.right && 
                             touch.clientY >= rect.top && 
                             touch.clientY <= rect.bottom;
      } else {
        touchStartedOnBoard = false;
      }
    };
    
    const handleTouchMove = (e) => {
      // 只在触摸开始于棋盘区域时阻止默认行为
      if (touchStartedOnBoard) {
        e.preventDefault();
      }
    };
    
    const handleTouchEnd = (e) => {
      if (!startX || !startY) return;
      
      // 只在触摸开始于棋盘区域时处理滑动
      if (!touchStartedOnBoard) {
        startX = startY = null;
        return;
      }
      
      const endX = e.changedTouches[0].clientX;
      const endY = e.changedTouches[0].clientY;
      
      const diffX = startX - endX;
      const diffY = startY - endY;
      
      // 检查滑动距离是否超过阈值
      if (Math.abs(diffX) < SWIPE_THRESHOLD && Math.abs(diffY) < SWIPE_THRESHOLD) {
        startX = startY = null;
        return; // 距离太短，视为点击而不是滑动
      }
      
      if (Math.abs(diffX) > Math.abs(diffY)) {
        if (diffX > 0) {
          this.move('left');
        } else {
          this.move('right');
        }
      } else {
        if (diffY > 0) {
          this.move('up');
        } else {
          this.move('down');
        }
      }
      
      startX = startY = null;
      touchStartedOnBoard = false;
    };
    document.addEventListener('touchstart', handleTouchStart, { passive: false });
    document.addEventListener('touchmove', handleTouchMove, { passive: false });
    document.addEventListener('touchend', handleTouchEnd, { passive: false });
    
    const restartButtons = document.querySelectorAll('#restart-button, #restart-button-over, #restart-button-won');
    restartButtons.forEach(button => {
      button.addEventListener('click', () => this.restart());
    });

    const undoButton = document.getElementById('undo-button');
    if (undoButton) {
      undoButton.addEventListener('click', () => this.undo());
    }
  }
}

(function() {
  let game2048Instance = null;
  let globalObserver = null;

  function checkAndInitGame(reset = false) {
    if (typeof window === 'undefined' || typeof document === 'undefined') return false;

    const gameContainer = document.getElementById('game2048');
    const gameBoard = document.getElementById('game-board-2048');

    // 如果游戏实例已经存在且不需要重置，直接返回
    if (game2048Instance && !reset) {
      return true;
    }

    if (gameContainer && gameBoard) {
      try {
        // 不恢复游戏状态，始终创建新游戏
        game2048Instance = new Game2048();
        return true;
      } catch (error) {
        return false;
      }
    }
    return false;
  }

  function setupDOMObserver() {
    const observer = new MutationObserver(function(mutationsList) {
      for (const mutation of mutationsList) {
        if (mutation.type === 'childList') {
          if (checkAndInitGame()) {
            observer.disconnect();
            return;
          }
        }
      }
    });

    observer.observe(document.body, {
      childList: true,
      subtree: true
    });

    return observer;
  }

  function handleRouteChange() {
    // 路由变化时重置游戏状态，确保用户返回时重新开始
    game2048Instance = null;
    // 延迟检查，确保 VuePress 有足够时间渲染页面
    setTimeout(function() {
      if (!checkAndInitGame(true)) { // 使用true参数强制重置游戏
        // 如果立即初始化失败，设置 DOM 观察器
        globalObserver = setupDOMObserver();
      }
    }, 1000);
  }

  // 检查是否在浏览器环境中
  function isBrowser() {
    return typeof window !== 'undefined' && typeof document !== 'undefined';
  }

  // 在多种情况下尝试初始化游戏
  if (isBrowser()) {
    let initialized = false;

    // 1. 立即尝试
    setTimeout(function() {
      if (!initialized && checkAndInitGame()) {
        initialized = true;
      }
    }, 200);

    // 2. DOMContentLoaded事件
    document.addEventListener('DOMContentLoaded', function() {
      setTimeout(function() {
        if (!initialized && checkAndInitGame()) {
          initialized = true;
        }
      }, 100);
    });

    // 3. window.load事件
    window.addEventListener('load', function() {
      setTimeout(function() {
        if (!initialized && checkAndInitGame()) {
          initialized = true;
        } else {
          // 如果load事件后仍未初始化，设置DOM观察器
          globalObserver = setupDOMObserver();
        }
      }, 50);
    });

    // 4. 2秒后再次尝试（作为备用）
    setTimeout(function() {
      if (!initialized && checkAndInitGame()) {
        initialized = true;
      } else {
        // 如果备用尝试也失败，设置DOM观察器
        globalObserver = setupDOMObserver();
      }
    }, 2000);

    // 5. 监听所有可能的页面显示/隐藏和路由变化事件
    document.addEventListener('visibilitychange', () => {
      if (document.visibilityState === 'visible') {
        setTimeout(function() {
          // 只有当游戏实例不存在时才初始化，否则保持现有游戏状态
          if (!game2048Instance) {
            if (!checkAndInitGame()) {
              // 如果立即初始化失败，设置DOM观察器
              globalObserver = setupDOMObserver();
            }
          } else {
            // 重新渲染棋盘，确保显示正确
            if (game2048Instance.render) {
              game2048Instance.render();
            }
          }
        }, 100);
      }
    });

    window.addEventListener('focus', () => {
      setTimeout(function() {
        // 只有当游戏实例不存在时才初始化，否则保持现有游戏状态
        if (!game2048Instance) {
          if (!checkAndInitGame()) {
            globalObserver = setupDOMObserver();
          }
        } else {
          // 重新渲染棋盘，确保显示正确
          if (game2048Instance.render) {
            game2048Instance.render();
          }
        }
      }, 100);
    });

    window.addEventListener('pageshow', () => {
      setTimeout(function() {
        if (!game2048Instance) {
          if (!checkAndInitGame()) {
            globalObserver = setupDOMObserver();
          }
        } else {
          if (game2048Instance.render) {
            game2048Instance.render();
          }
        }
      }, 100);
    });

    // 设置路由变化监听器
    window.addEventListener('hashchange', handleRouteChange);
    window.addEventListener('popstate', handleRouteChange);
  }
})();
</script>
